# /etc/systemd/system/cex-engine.service
[Unit]
Description=CEX Trading Engine (Rust)
After=network.target redis6.service
Wants=redis6.service

[Service]
Type=simple
User=ec2-user
WorkingDirectory=/home/ec2-user/hustlecoin-cex/rust-engine
# 启动前清掉任何不在本 service cgroup 里的孤儿实例(防多实例覆盖写 spreads)。
# 前缀 '-' = 忽略退出码(pkill 无匹配返回 1);ExecStartPre 在 ExecStart 之前跑,只命中孤儿。
ExecStartPre=-/usr/bin/pkill -9 -f target/release/cex-engine
ExecStart=/home/ec2-user/hustlecoin-cex/rust-engine/target/release/cex-engine
Environment=RUST_LOG=cex_engine=info
Environment=REDIS_URL=redis://127.0.0.1:6379
Restart=always
RestartSec=3
LimitNOFILE=1000000

[Install]
WantedBy=multi-user.target
